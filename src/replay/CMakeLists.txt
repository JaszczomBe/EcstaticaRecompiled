if(NOT E2R_ENABLE_RUNTIME_OPTIONS)
  add_library(e2r_replay STATIC src/ReplayStub.c)
  target_include_directories(e2r_replay PUBLIC include)
  return()
endif()
add_library(e2r_replay STATIC src/Replay.cpp)
target_include_directories(e2r_replay PUBLIC include)
target_compile_features(e2r_replay PRIVATE cxx_std_20)
target_compile_definitions(e2r_replay PRIVATE _FILE_OFFSET_BITS=64)
target_compile_definitions(e2r_replay PRIVATE E2R_REPLAY_BACKEND="${E2R_HOST_BACKEND}")
target_compile_options(e2r_replay PRIVATE -Wall -Wextra -O2)
if(E2R_ENABLE_ASAN)
  target_compile_definitions(e2r_replay PRIVATE E2R_REPLAY_ASAN=1)
endif()
target_link_libraries(e2r_replay PUBLIC e2r_artifacts)
add_custom_target(e2r-replay-identity
  COMMAND ${CMAKE_COMMAND} -DSOURCE=${CMAKE_SOURCE_DIR} -DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/ReplayIdentity.h -P ${CMAKE_SOURCE_DIR}/cmake/E2RBuildIdentity.cmake
  BYPRODUCTS ${CMAKE_CURRENT_BINARY_DIR}/ReplayIdentity.h)
add_dependencies(e2r_replay e2r-replay-identity)
target_include_directories(e2r_replay PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
option(E2R_REPLAY_BUILD_TESTS "Build replay clock/order/recovery contracts" OFF)
if(E2R_REPLAY_BUILD_TESTS)
  add_executable(e2r-replay-contract tests/contract.cpp)
  target_link_libraries(e2r-replay-contract PRIVATE e2r_replay)
  if(E2R_ENABLE_ASAN)
    target_compile_options(e2r-replay-contract PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(e2r-replay-contract PRIVATE -fsanitize=address)
  endif()
endif()
